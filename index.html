// Предполагается, что объект supabase уже инициализирован в вашем проекте: // const supabase = supabase.createClient('YOUR_SUPABASE_URL', 'YOUR_SUPABASE_ANON_KEY'); /** * Функция 1: Сохранение накопленного времени учебной сессии в базу данных * @param {string} userId - Уникальный ID студента (Романа) * @param {number} secondsSpent - Сколько секунд длилась текущая сессия */ async function saveSession(userId, secondsSpent) { try { // Отправляем запрос на вставку новой строки в таблицу учебных сессий const { data, error } = await supabase .from('study_sessions') .insert([ { user_id: userId, duration_seconds: secondsSpent, ended_at: new Date().toISOString() // Текущее время завершения } ]); if (error) throw error; console.log('Сессия успешно сохранена в Supabase:', data); return true; } catch (error) { console.error('Ошибка при сохранении сессии:', error.message); return false; } } /** * Функция 2: Обновление статуса активности (онлайн/перерыв/офлайн) в реальном времени * @param {string} userId - Уникальный ID студента * @param {string} currentStatus - Строка статуса ('online', 'pause', 'offline') */ async function updateOnlineStatus(userId, currentStatus) { try { // Обновляем поле status в таблице профилей пользователей const { data, error } = await supabase .from('profiles') .update({ status: currentStatus, last_seen: new Date().toISOString() // Фиксируем время последней активности }) .eq('id', userId); // Фильтр: обновляем только строку текущего пользователя if (error) throw error; console.log(`Статус пользователя изменен на "${currentStatus}"`); return true; } catch (error) { console.error('Ошибка обновления статуса:', error.message); return false; } }
